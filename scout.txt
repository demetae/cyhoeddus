[CmdletBinding()]
param(
    [string] $ScoutExe = (Join-Path $PSScriptRoot 'Scripts\scout.exe'),

    # Alternative to manually entering credentials.
    [string] $AwsProfile,

    # Prompt securely for AWS credentials each time the script runs.
    [switch] $PromptForAwsCredentials
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

function ConvertFrom-SecureValue {
    param(
        [Parameter(Mandatory)]
        [Security.SecureString] $SecureValue
    )

    $pointer = [Runtime.InteropServices.Marshal]::SecureStringToBSTR(
        $SecureValue
    )

    try {
        return [Runtime.InteropServices.Marshal]::PtrToStringBSTR(
            $pointer
        )
    }
    finally {
        [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($pointer)
    }
}

function ConvertTo-NativeArgument {
    param(
        [Parameter(Mandatory)]
        [AllowEmptyString()]
        [string] $Argument
    )

    if ($Argument.Length -eq 0) {
        return '""'
    }

    if ($Argument -notmatch '[\s"]') {
        return $Argument
    }

    $escaped = [regex]::Replace(
        $Argument,
        '(\\*)"',
        '$1$1\"'
    )

    $escaped = [regex]::Replace(
        $escaped,
        '(\\+)$',
        '$1$1'
    )

    return '"' + $escaped + '"'
}

function Invoke-NativeProcess {
    param(
        [Parameter(Mandatory)]
        [string] $FilePath,

        [string[]] $Arguments = @(),

        [hashtable] $EnvironmentVariables = @{}
    )

    if (-not (Test-Path -LiteralPath $FilePath -PathType Leaf)) {
        throw "Executable not found: $FilePath"
    }

    $startInfo = New-Object System.Diagnostics.ProcessStartInfo

    $startInfo.FileName = (
        Resolve-Path -LiteralPath $FilePath
    ).Path

    $startInfo.Arguments = (
        $Arguments |
            ForEach-Object {
                ConvertTo-NativeArgument -Argument $_
            }
    ) -join ' '

    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true

    foreach ($name in $EnvironmentVariables.Keys) {
        $value = $EnvironmentVariables[$name]

        if ([string]::IsNullOrEmpty($value)) {
            $startInfo.EnvironmentVariables.Remove($name)
        }
        else {
            $startInfo.EnvironmentVariables[$name] = $value
        }
    }

    $process = New-Object System.Diagnostics.Process
    $process.StartInfo = $startInfo

    try {
        if (-not $process.Start()) {
            throw "Failed to start native process: $FilePath"
        }

        # Read both streams concurrently to avoid buffer deadlocks.
        $stdoutTask = $process.StandardOutput.ReadToEndAsync()
        $stderrTask = $process.StandardError.ReadToEndAsync()

        $process.WaitForExit()

        return [pscustomobject]@{
            ExitCode = $process.ExitCode
            StdOut   = $stdoutTask.GetAwaiter().GetResult()
            StdErr   = $stderrTask.GetAwaiter().GetResult()
        }
    }
    finally {
        $process.Dispose()
    }
}

function Get-ScoutSuiteServiceList {
    param(
        [Parameter(Mandatory)]
        [string] $Output
    )

    $marker = 'The available services are:'

    $markerIndex = $Output.IndexOf(
        $marker,
        [System.StringComparison]::OrdinalIgnoreCase
    )

    if ($markerIndex -lt 0) {
        throw @"
ScoutSuite completed successfully, but the expected service-list marker
was not found.

Expected marker:
$marker
"@
    }

    $serviceText = $Output.Substring(
        $markerIndex + $marker.Length
    )

    # ScoutSuite emits a quoted comma-separated list rather than a JSON array:
    #
    # "acm", "awslambda", "cloudformation", ...
    #
    # Extract each quoted service name directly.
    $matches = [regex]::Matches(
        $serviceText,
        '"([^"]+)"'
    )

    $services = @(
        foreach ($match in $matches) {
            $service = $match.Groups[1].Value.Trim()

            if (
                -not [string]::IsNullOrWhiteSpace($service) -and
                $service -match '^[A-Za-z0-9_-]+$'
            ) {
                $service
            }
        }
    )

    if ($services.Count -eq 0) {
        throw @"
ScoutSuite completed successfully, but no quoted service names could be
extracted from its output.

Text following the service-list marker:
$serviceText
"@
    }

    return $services
}

if ($AwsProfile -and $PromptForAwsCredentials) {
    throw 'Choose either -AwsProfile or -PromptForAwsCredentials, not both.'
}

$scoutEnvironment = @{}

if ($PromptForAwsCredentials) {
    Write-Host ''
    Write-Host 'Enter AWS credentials for this ScoutSuite run.'
    Write-Host 'Secret values will not be shown while typing.'
    Write-Host ''

    $accessKeyId = Read-Host 'AWS access key ID'

    if ([string]::IsNullOrWhiteSpace($accessKeyId)) {
        throw 'AWS access key ID cannot be blank.'
    }

    $secretAccessKeySecure = Read-Host `
        'AWS secret access key' `
        -AsSecureString

    $sessionTokenSecure = Read-Host `
        'AWS session token - press Enter if your credentials do not use one' `
        -AsSecureString

    $secretAccessKey = ConvertFrom-SecureValue `
        -SecureValue $secretAccessKeySecure

    $sessionToken = ConvertFrom-SecureValue `
        -SecureValue $sessionTokenSecure

    if ([string]::IsNullOrWhiteSpace($secretAccessKey)) {
        throw 'AWS secret access key cannot be blank.'
    }

    $scoutEnvironment['AWS_ACCESS_KEY_ID'] = $accessKeyId
    $scoutEnvironment['AWS_SECRET_ACCESS_KEY'] = $secretAccessKey

    if ([string]::IsNullOrWhiteSpace($sessionToken)) {
        # Clear inherited token values when using long-lived keys.
        $scoutEnvironment['AWS_SESSION_TOKEN'] = ''
        $scoutEnvironment['AWS_SECURITY_TOKEN'] = ''
    }
    else {
        $scoutEnvironment['AWS_SESSION_TOKEN'] = $sessionToken
        $scoutEnvironment['AWS_SECURITY_TOKEN'] = $sessionToken
    }
}

Write-Host ''
Write-Host 'Discovering supported ScoutSuite services...'

$scoutArguments = @('aws')

if (-not [string]::IsNullOrWhiteSpace($AwsProfile)) {
    $scoutArguments += @(
        '--profile',
        $AwsProfile
    )
}

$scoutArguments += '--list-services'

try {
    $result = Invoke-NativeProcess `
        -FilePath $ScoutExe `
        -Arguments $scoutArguments `
        -EnvironmentVariables $scoutEnvironment
}
finally {
    # Clear plaintext credential variables after ScoutSuite exits.
    if (Get-Variable accessKeyId -ErrorAction SilentlyContinue) {
        $accessKeyId = $null
    }

    if (Get-Variable secretAccessKey -ErrorAction SilentlyContinue) {
        $secretAccessKey = $null
    }

    if (Get-Variable sessionToken -ErrorAction SilentlyContinue) {
        $sessionToken = $null
    }

    $scoutEnvironment.Clear()
}

# ScoutSuite writes normal informational logs to stderr.
if (-not [string]::IsNullOrWhiteSpace($result.StdErr)) {
    foreach ($line in ($result.StdErr -split "\r?\n")) {
        if (-not [string]::IsNullOrWhiteSpace($line)) {
            Write-Host "[ScoutSuite] $line" -ForegroundColor DarkGray
        }
    }
}

# Non-zero exit codes are genuine ScoutSuite failures.
if ($result.ExitCode -ne 0) {
    $details = $result.StdErr.Trim()

    if ([string]::IsNullOrWhiteSpace($details)) {
        $details = $result.StdOut.Trim()
    }

    throw @"
ScoutSuite service discovery failed.
Executable: $ScoutExe
Exit code:  $($result.ExitCode)

$details
"@
}

# Search both streams because ScoutSuite versions may differ in where
# informational messages are written.
$combinedOutput = @(
    $result.StdOut
    $result.StdErr
) -join "`n"

$supportedServices = @(
    Get-ScoutSuiteServiceList -Output $combinedOutput
)

if ($supportedServices.Count -eq 0) {
    throw 'ScoutSuite completed successfully but reported no services.'
}

Write-Host ''
Write-Host "ScoutSuite reported $($supportedServices.Count) supported services:"
Write-Host ''

$supportedServices |
    Sort-Object -Unique |
    ForEach-Object {
        Write-Host "  $_"
    }

return $supportedServices
