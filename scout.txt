[CmdletBinding()]
param(
    # When this script is stored in the ScoutSuite virtual-environment folder,
    # the default resolves to .\Scripts\scout.exe
    [string] $ScoutExe = (Join-Path $PSScriptRoot 'Scripts\scout.exe'),

    # Optional AWS CLI / boto3 profile name.
    [string] $AwsProfile
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

function ConvertTo-NativeArgument {
    param(
        [Parameter(Mandatory)]
        [AllowEmptyString()]
        [string] $Argument
    )

    # Safely quote arguments for ProcessStartInfo.Arguments.
    # This supports Windows PowerShell 5.1 as well as PowerShell 7.
    if ($Argument.Length -eq 0) {
        return '""'
    }

    if ($Argument -notmatch '[\s"]') {
        return $Argument
    }

    $escaped = [regex]::Replace($Argument, '(\\*)"', '$1$1\"')
    $escaped = [regex]::Replace($escaped, '(\\+)$', '$1$1')

    return '"' + $escaped + '"'
}

function Invoke-NativeProcess {
    param(
        [Parameter(Mandatory)]
        [string] $FilePath,

        [Parameter()]
        [string[]] $Arguments = @()
    )

    if (-not (Test-Path -LiteralPath $FilePath -PathType Leaf)) {
        throw "Executable not found: $FilePath"
    }

    $startInfo = New-Object System.Diagnostics.ProcessStartInfo
    $startInfo.FileName = (Resolve-Path -LiteralPath $FilePath).Path
    $startInfo.Arguments = (
        $Arguments |
            ForEach-Object { ConvertTo-NativeArgument -Argument $_ }
    ) -join ' '

    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true

    $process = New-Object System.Diagnostics.Process
    $process.StartInfo = $startInfo

    try {
        if (-not $process.Start()) {
            throw "Failed to start native process: $FilePath"
        }

        # Start reading both streams before waiting for the process.
        # This avoids deadlocks when either buffer fills up.
        $stdoutTask = $process.StandardOutput.ReadToEndAsync()
        $stderrTask = $process.StandardError.ReadToEndAsync()

        $process.WaitForExit()

        $stdout = $stdoutTask.GetAwaiter().GetResult()
        $stderr = $stderrTask.GetAwaiter().GetResult()

        return [pscustomobject]@{
            ExitCode = $process.ExitCode
            StdOut   = $stdout
            StdErr   = $stderr
        }
    }
    finally {
        $process.Dispose()
    }
}

Write-Host 'Discovering supported ScoutSuite services...'

$scoutArguments = @('aws')

if (-not [string]::IsNullOrWhiteSpace($AwsProfile)) {
    $scoutArguments += @('--profile', $AwsProfile)
}

$scoutArguments += '--list-services'

$result = Invoke-NativeProcess `
    -FilePath $ScoutExe `
    -Arguments $scoutArguments

# ScoutSuite may write normal informational logs to stderr.
# Display them without turning them into PowerShell errors.
if (-not [string]::IsNullOrWhiteSpace($result.StdErr)) {
    foreach ($line in ($result.StdErr -split "\r?\n")) {
        if (-not [string]::IsNullOrWhiteSpace($line)) {
            Write-Host "[ScoutSuite] $line" -ForegroundColor DarkGray
        }
    }
}

if ($result.ExitCode -ne 0) {
    $details = $result.StdErr.Trim()

    if ([string]::IsNullOrWhiteSpace($details)) {
        $details = $result.StdOut.Trim()
    }

    throw @"
ScoutSuite service discovery failed.
Executable: $ScoutExe
Exit code:  $($result.ExitCode)

$details
"@
}

$SupportedServices = @(
    $result.StdOut -split "\r?\n" |
        ForEach-Object { $_.Trim() } |
        Where-Object { -not [string]::IsNullOrWhiteSpace($_) }
)

if ($SupportedServices.Count -eq 0) {
    throw @"
ScoutSuite completed successfully but returned no services on stdout.

ScoutSuite stderr:
$($result.StdErr.Trim())
"@
}

Write-Host ''
Write-Host "ScoutSuite reported $($SupportedServices.Count) supported service entries:"
Write-Host ''

$SupportedServices | ForEach-Object {
    Write-Host "  $_"
}

# Leave the discovered values available to calling code.
return $SupportedServices
