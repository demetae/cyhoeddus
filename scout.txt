[CmdletBinding()]
param(
    [string] $ScoutExe = (Join-Path $PSScriptRoot 'Scripts\scout.exe'),

    # Alternative to manually entering credentials.
    [string] $AwsProfile,

    # Prompt securely for AWS credentials each time the script runs.
    [switch] $PromptForAwsCredentials,

    # Only scan these authorised AWS regions.
    [string[]] $Regions = @(
        'eu-west-2',
        'eu-central-1'
    ),

    # Known SCP-blocked services from the earlier ScoutSuite run.
    [string[]] $ExcludedServices = @(
        'cloudfront',
        'directconnect'
    ),

    # Optionally scan only specified services.
    # Leave empty to discover and scan all supported services.
    [string[]] $Services,

    # Stop a single service scan if it runs for too long.
    # Set to 0 to disable the timeout.
    [int] $PerServiceTimeoutMinutes = 45,

    # Include ScoutSuite stack traces in the logs.
    [switch] $DebugLogging
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

# -------------------------------------------------------------------
# Helper functions
# -------------------------------------------------------------------

function ConvertFrom-SecureValue {
    param(
        [Parameter(Mandatory)]
        [Security.SecureString] $SecureValue
    )

    $pointer = [Runtime.InteropServices.Marshal]::SecureStringToBSTR(
        $SecureValue
    )

    try {
        return [Runtime.InteropServices.Marshal]::PtrToStringBSTR(
            $pointer
        )
    }
    finally {
        if ($pointer -ne [IntPtr]::Zero) {
            [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($pointer)
        }
    }
}

function ConvertTo-NativeArgument {
    param(
        [Parameter(Mandatory)]
        [AllowEmptyString()]
        [string] $Argument
    )

    if ($Argument.Length -eq 0) {
        return '""'
    }

    if ($Argument -notmatch '[\s"]') {
        return $Argument
    }

    $escaped = [regex]::Replace(
        $Argument,
        '(\\*)"',
        '$1$1\"'
    )

    $escaped = [regex]::Replace(
        $escaped,
        '(\\+)$',
        '$1$1'
    )

    return '"' + $escaped + '"'
}

function Invoke-NativeProcess {
    param(
        [Parameter(Mandatory)]
        [string] $FilePath,

        [string[]] $Arguments = @(),

        [hashtable] $EnvironmentVariables = @{},

        [int] $TimeoutSeconds = 0
    )

    if (-not (Test-Path -LiteralPath $FilePath -PathType Leaf)) {
        throw "Executable not found: $FilePath"
    }

    $startInfo = New-Object System.Diagnostics.ProcessStartInfo

    $startInfo.FileName = (
        Resolve-Path -LiteralPath $FilePath
    ).Path

    $startInfo.Arguments = (
        $Arguments |
            ForEach-Object {
                ConvertTo-NativeArgument -Argument $_
            }
    ) -join ' '

    $startInfo.UseShellExecute = $false
    $startInfo.CreateNoWindow = $true
    $startInfo.RedirectStandardOutput = $true
    $startInfo.RedirectStandardError = $true

    # Existing environment variables, including HTTPS_PROXY, are inherited.
    # Add or remove only the values explicitly supplied here.
    foreach ($name in $EnvironmentVariables.Keys) {
        $value = $EnvironmentVariables[$name]

        if ([string]::IsNullOrEmpty($value)) {
            $startInfo.EnvironmentVariables.Remove($name)
        }
        else {
            $startInfo.EnvironmentVariables[$name] = $value
        }
    }

    $process = New-Object System.Diagnostics.Process
    $process.StartInfo = $startInfo

    try {
        if (-not $process.Start()) {
            throw "Failed to start native process: $FilePath"
        }

        # Read both streams concurrently to avoid buffer deadlocks.
        $stdoutTask = $process.StandardOutput.ReadToEndAsync()
        $stderrTask = $process.StandardError.ReadToEndAsync()

        $timedOut = $false

        if ($TimeoutSeconds -gt 0) {
            $completed = $process.WaitForExit($TimeoutSeconds * 1000)

            if (-not $completed) {
                $timedOut = $true

                try {
                    # Supported by newer .NET runtimes.
                    $process.Kill($true)
                }
                catch {
                    # Fallback for Windows PowerShell 5.1.
                    $process.Kill()
                }

                $process.WaitForExit()
            }
        }
        else {
            $process.WaitForExit()
        }

        return [pscustomobject]@{
            ExitCode = if ($timedOut) { $null } else { $process.ExitCode }
            TimedOut = $timedOut
            StdOut   = $stdoutTask.GetAwaiter().GetResult()
            StdErr   = $stderrTask.GetAwaiter().GetResult()
        }
    }
    finally {
        $process.Dispose()
    }
}

function Get-ScoutSuiteServiceList {
    param(
        [Parameter(Mandatory)]
        [string] $Output
    )

    $marker = 'The available services are:'

    $markerIndex = $Output.IndexOf(
        $marker,
        [System.StringComparison]::OrdinalIgnoreCase
    )

    if ($markerIndex -lt 0) {
        throw @"
ScoutSuite completed successfully, but the expected service-list marker
was not found.

Expected marker:
$marker
"@
    }

    $serviceText = $Output.Substring(
        $markerIndex + $marker.Length
    )

    $matches = [regex]::Matches(
        $serviceText,
        '"([^"]+)"'
    )

    $servicesFound = @(
        foreach ($match in $matches) {
            $service = $match.Groups[1].Value.Trim()

            if (
                -not [string]::IsNullOrWhiteSpace($service) -and
                $service -match '^[A-Za-z0-9_-]+$'
            ) {
                $service
            }
        }
    )

    if ($servicesFound.Count -eq 0) {
        throw @"
ScoutSuite completed successfully, but no quoted service names could be
extracted from its output.

Text following the service-list marker:
$serviceText
"@
    }

    return $servicesFound | Sort-Object -Unique
}

# -------------------------------------------------------------------
# Validate arguments
# -------------------------------------------------------------------

if ($AwsProfile -and $PromptForAwsCredentials) {
    throw 'Choose either -AwsProfile or -PromptForAwsCredentials, not both.'
}

if (-not (Test-Path -LiteralPath $ScoutExe -PathType Leaf)) {
    throw "ScoutSuite executable not found: $ScoutExe"
}

# -------------------------------------------------------------------
# Prepare credentials
# -------------------------------------------------------------------

$scoutEnvironment = @{}

$accessKeyId = $null
$secretAccessKey = $null
$sessionToken = $null

if ($PromptForAwsCredentials) {
    Write-Host ''
    Write-Host 'Enter AWS credentials for this ScoutSuite run.'
    Write-Host 'Secret values will not be shown while typing.'
    Write-Host ''

    $accessKeyId = Read-Host 'AWS access key ID'

    if ([string]::IsNullOrWhiteSpace($accessKeyId)) {
        throw 'AWS access key ID cannot be blank.'
    }

    $secretAccessKeySecure = Read-Host `
        'AWS secret access key' `
        -AsSecureString

    $sessionTokenSecure = Read-Host `
        'AWS session token - press Enter if your credentials do not use one' `
        -AsSecureString

    $secretAccessKey = ConvertFrom-SecureValue `
        -SecureValue $secretAccessKeySecure

    $sessionToken = ConvertFrom-SecureValue `
        -SecureValue $sessionTokenSecure

    if ([string]::IsNullOrWhiteSpace($secretAccessKey)) {
        throw 'AWS secret access key cannot be blank.'
    }

    $scoutEnvironment['AWS_ACCESS_KEY_ID'] = $accessKeyId
    $scoutEnvironment['AWS_SECRET_ACCESS_KEY'] = $secretAccessKey

    if ([string]::IsNullOrWhiteSpace($sessionToken)) {
        # Clear inherited token values when using long-lived keys.
        $scoutEnvironment['AWS_SESSION_TOKEN'] = ''
        $scoutEnvironment['AWS_SECURITY_TOKEN'] = ''
    }
    else {
        $scoutEnvironment['AWS_SESSION_TOKEN'] = $sessionToken
        $scoutEnvironment['AWS_SECURITY_TOKEN'] = $sessionToken
    }
}

# -------------------------------------------------------------------
# Prepare the report directory
# -------------------------------------------------------------------

$timestamp = Get-Date -Format 'yyyyMMdd-HHmmss'

$rootDir = Join-Path `
    (Get-Location) `
    "scoutsuite-per-service-$timestamp"

New-Item `
    -ItemType Directory `
    -Path $rootDir `
    -Force |
    Out-Null

$summaryFile = Join-Path $rootDir 'summary.csv'

Write-Host ''
Write-Host 'ScoutSuite per-service scan'
Write-Host "Output directory: $rootDir"
Write-Host "Regions:          $($Regions -join ', ')"
Write-Host "Excluded:         $($ExcludedServices -join ', ')"
Write-Host ''

# -------------------------------------------------------------------
# Discover supported services unless explicitly supplied
# -------------------------------------------------------------------

try {
    if (-not $Services -or $Services.Count -eq 0) {
        Write-Host 'Discovering supported ScoutSuite services...'

        $discoveryArguments = @('aws')

        if (-not [string]::IsNullOrWhiteSpace($AwsProfile)) {
            $discoveryArguments += @(
                '--profile',
                $AwsProfile
            )
        }

        $discoveryArguments += '--list-services'

        $discoveryResult = Invoke-NativeProcess `
            -FilePath $ScoutExe `
            -Arguments $discoveryArguments `
            -EnvironmentVariables $scoutEnvironment

        # ScoutSuite writes normal informational logs to stderr.
        if (-not [string]::IsNullOrWhiteSpace($discoveryResult.StdErr)) {
            foreach ($line in ($discoveryResult.StdErr -split "\r?\n")) {
                if (-not [string]::IsNullOrWhiteSpace($line)) {
                    Write-Host "[ScoutSuite] $line" -ForegroundColor DarkGray
                }
            }
        }

        if ($discoveryResult.ExitCode -ne 0) {
            $details = $discoveryResult.StdErr.Trim()

            if ([string]::IsNullOrWhiteSpace($details)) {
                $details = $discoveryResult.StdOut.Trim()
            }

            throw @"
ScoutSuite service discovery failed.
Executable: $ScoutExe
Exit code:  $($discoveryResult.ExitCode)

$details
"@
        }

        $combinedOutput = @(
            $discoveryResult.StdOut
            $discoveryResult.StdErr
        ) -join "`n"

        $Services = @(
            Get-ScoutSuiteServiceList -Output $combinedOutput
        )
    }

    $Services = @(
        $Services |
            Where-Object {
                $_ -and $_ -notin $ExcludedServices
            } |
            Sort-Object -Unique
    )

    if ($Services.Count -eq 0) {
        throw 'No ScoutSuite services remain after applying exclusions.'
    }

    Write-Host ''
    Write-Host "Services to scan: $($Services.Count)"
    Write-Host ''

    foreach ($service in $Services) {
        Write-Host "  $service"
    }

    # -------------------------------------------------------------------
    # Scan each service independently
    # -------------------------------------------------------------------

    $results = [System.Collections.Generic.List[object]]::new()

    foreach ($service in $Services) {
        $serviceDir = Join-Path $rootDir $service

        New-Item `
            -ItemType Directory `
            -Path $serviceDir `
            -Force |
            Out-Null

        $stdoutLog = Join-Path $serviceDir 'stdout.log'
        $stderrLog = Join-Path $serviceDir 'stderr.log'
        $scoutLog  = Join-Path $serviceDir 'scout.log'

        $reportName = "aws-$service"

        $arguments = @(
            'aws'
        )

        if (-not [string]::IsNullOrWhiteSpace($AwsProfile)) {
            $arguments += @(
                '--profile',
                $AwsProfile
            )
        }

        $arguments += @(
            '--regions'
        )

        $arguments += $Regions

        $arguments += @(
            '--services',
            $service,
            '--report-dir',
            $serviceDir,
            '--report-name',
            $reportName,
            '--logfile',
            $scoutLog,
            '--no-browser',
            '--force'
        )

        if ($DebugLogging) {
            $arguments += '--debug'
        }

        $startTime = Get-Date

        Write-Host ''
        Write-Host ('=' * 72)
        Write-Host "Starting service: $service"
        Write-Host "Started:          $startTime"
        Write-Host "Regions:          $($Regions -join ', ')"
        Write-Host "Report folder:    $serviceDir"
        Write-Host "ScoutSuite log:   $scoutLog"
        Write-Host ('=' * 72)

        $timeoutSeconds = if ($PerServiceTimeoutMinutes -gt 0) {
            $PerServiceTimeoutMinutes * 60
        }
        else {
            0
        }

        $scanResult = Invoke-NativeProcess `
            -FilePath $ScoutExe `
            -Arguments $arguments `
            -EnvironmentVariables $scoutEnvironment `
            -TimeoutSeconds $timeoutSeconds

        Set-Content `
            -LiteralPath $stdoutLog `
            -Value $scanResult.StdOut `
            -Encoding UTF8

        Set-Content `
            -LiteralPath $stderrLog `
            -Value $scanResult.StdErr `
            -Encoding UTF8

        $endTime = Get-Date
        $duration = $endTime - $startTime

        $status = if ($scanResult.TimedOut) {
            'TimedOut'
        }
        elseif ($scanResult.ExitCode -eq 0) {
            'Completed'
        }
        else {
            'CompletedWithErrors'
        }

        $results.Add(
            [pscustomobject]@{
                Service         = $service
                Status          = $status
                ExitCode        = $scanResult.ExitCode
                Started         = $startTime
                Finished        = $endTime
                DurationMinutes = [Math]::Round(
                    $duration.TotalMinutes,
                    2
                )
                ReportDirectory = $serviceDir
                ScoutLog        = $scoutLog
                StandardOutput  = $stdoutLog
                StandardError   = $stderrLog
            }
        )

        # Update after each service so partial progress is preserved.
        $results |
            Export-Csv `
                -LiteralPath $summaryFile `
                -NoTypeInformation `
                -Force

        Write-Host ''
        Write-Host "Finished service: $service"
        Write-Host "Status:           $status"
        Write-Host "Exit code:        $($scanResult.ExitCode)"
        Write-Host "Duration:         $([Math]::Round($duration.TotalMinutes, 2)) minutes"

        if ($scanResult.TimedOut) {
            Write-Host `
                "The scan exceeded $PerServiceTimeoutMinutes minutes and was stopped." `
                -ForegroundColor Yellow
        }
    }

    Write-Host ''
    Write-Host ('=' * 72)
    Write-Host 'All requested per-service scans have finished.'
    Write-Host "Summary file: $summaryFile"
    Write-Host ('=' * 72)
}
finally {
    # Clear plaintext credential values after all service scans complete.
    $accessKeyId = $null
    $secretAccessKey = $null
    $sessionToken = $null

    if ($scoutEnvironment) {
        $scoutEnvironment.Clear()
    }
}
