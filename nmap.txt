files=(
  targets_*.txt(N)
  .txt(N)
)

for file in $files; do
  output="${file%.txt}"
  normal_output="${output}.nmap"
  ports="${output##*_}"

  # Skip completed scans and resume interrupted scans.
  if [[ -e "$normal_output" ]]; then
    if grep -q '^# Nmap done at ' -- "$normal_output"; then
      print -- "Skipping completed scan: $normal_output"
      continue
    fi

    print -- "Resuming incomplete scan: $normal_output"
    sudo nmap --resume "$normal_output"
    continue
  fi

  if [[ "$ports" =~ '^([0-9]+)(-([0-9]+))?$' ]]; then
    first_port="${match[1]}"
    last_port="${match[3]:-$first_port}"

    if (( first_port >= 1 &&
          last_port <= 65535 &&
          first_port <= last_port )); then

      print -- "Scanning CIDRs from $file on TCP port(s) $ports"

      sudo nmap \
        -sS \
        -n \
        -Pn \
        -iL "$file" \
        -p "$ports" \
        --open \
        -vv \
        --max-retries 2 \
        --max-rtt-timeout 1000ms \
        -oA "$output"

      continue
    fi
  fi

  print -u2 -- "Skipping $file: filename does not end with a valid TCP port or range"
done
